cmake_minimum_required(VERSION 3.20)
project(CrispyCore VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(CrispyCore STATIC src/core.cpp src/gb.cpp src/gba.cpp)
target_include_directories(CrispyCore PUBLIC include)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(CrispyCore PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(crispy-core tools/main.cpp)
target_link_libraries(crispy-core PRIVATE CrispyCore)
include(CTest)
if(BUILD_TESTING)
  add_executable(crispy-core-checks tests/core_checks.cpp)
  target_link_libraries(crispy-core-checks PRIVATE CrispyCore)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_custom_target(core-demos ALL
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/demos.py ${CMAKE_CURRENT_BINARY_DIR}/demos)
  add_dependencies(crispy-core-checks core-demos)
  add_test(NAME core-checks COMMAND crispy-core-checks ${CMAKE_CURRENT_BINARY_DIR}/demos)
endif()
